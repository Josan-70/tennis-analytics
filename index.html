let bleDevice = null; let gattServer = null; let isRecording = false; let sessionShots = []; let autoReconnectTimer = null; // BLE接続メイン処理 async function connectBLE() { try { bleDevice = await navigator.bluetooth.requestDevice({ filters: [{ name: 'XIAO-Tennis' }], optionalServices: [ '19b10000-e8f2-537e-4f6c-d104768a1214', '0000180f-0000-1000-8000-00805f9b34fb' ] }); // 切断イベントの監視登録（自動再接続のフック） bleDevice.addEventListener('gattserverdisconnected', onDisconnected); await startGattConnection(); } catch (error) { console.error("BLE Connection Error:", error); } } // GATTサーバーへの接続処理 async function startGattConnection() { if (!bleDevice) return; updateStatus("接続中..."); gattServer = await bleDevice.gatt.connect(); // テニスデータサービス登録 const tennisService = await gattServer.getPrimaryService('19b10000-e8f2-537e-4f6c-d104768a1214'); const dataChar = await tennisService.getCharacteristic('19b10001-e8f2-537e-4f6c-d104768a1214'); await dataChar.startNotifications(); dataChar.addEventListener('characteristicvaluechanged', handleData); // バッテリーサービス登録 try { const battService = await gattServer.getPrimaryService('0000180f-0000-1000-8000-00805f9b34fb'); const battChar = await battService.getCharacteristic('00002a19-0000-1000-8000-00805f9b34fb'); await battChar.startNotifications(); battChar.addEventListener('characteristicvaluechanged', handleBatteryData); } catch(e) { console.log("Battery service skip"); } updateStatus("接続完了（計測中）"); if (autoReconnectTimer) clearInterval(autoReconnectTimer); } // 【機能1】電波切断時の自動再接続処理 function onDisconnected(event) { updateStatus("⚠️ 電波切断：再接続を試みています..."); // 3秒おきにバックグラウンドで自動再接続をリトライ if (autoReconnectTimer) clearInterval(autoReconnectTimer); autoReconnectTimer = setInterval(async () => { if (bleDevice && !bleDevice.gatt.connected) { try { await startGattConnection(); } catch (err) { console.log("再接続リトライ中..."); } } }, 3000); } // 【機能2】データ解析とリアルタイム集計（MAX/AVE/内訳） function handleData(event) { const value = event.target.value; const dataView = new DataView(value.buffer); // Float32 × 9要素のデコード const accelMag = dataView.getFloat32(0, true); const speed = dataView.getFloat32(4, true); const spin = dataView.getFloat32(8, true); const gx = dataView.getFloat32(24, true); const gz = dataView.getFloat32(32, true); // ショット判別 let shotType = "FOREHAND"; if (gz > 150) shotType = "SERVE"; else if (gx < -100) shotType = "BACKHAND"; else if (accelMag > 5.0 && gz > 100) shotType = "SMASH"; const shotData = { speed: speed, spin: spin, type: shotType, accel: accelMag }; // パネルの即時表示更新 updateCurrentPanel(shotData); // 記録中の場合、配列保持＆MAX/AVE/内訳の再計算を実行 if (isRecording) { sessionShots.push(shotData); calculateSessionStats(sessionShots); } } // バッテリーデータ解析 (mV ➔ %) function handleBatteryData(event) { const mv = event.target.value.getUint16(0, true); // 4200mV (100%) ~ 3300mV (0%) 計算 let pct = Math.round(((mv - 3300) / (4200 - 3300)) * 100); if (pct > 100) pct = 100; if (pct < 0) pct = 0; document.getElementById('batt-mv').textContent = mv; document.getElementById('batt-pct').textContent = pct + "%"; } // 集計ロジック（MAX / AVE / F,B,S,Sm） function calculateSessionStats(shots) { if (shots.length === 0) return; let maxSpeed = 0, sumSpeed = 0; let maxSpin = 0, sumSpin = 0; let counts = { FOREHAND: 0, BACKHAND: 0, SERVE: 0, SMASH: 0 }; shots.forEach(s => { if (s.speed > maxSpeed) maxSpeed = s.speed; sumSpeed += s.speed; if (s.spin > maxSpin) maxSpin = s.spin; sumSpin += s.spin; if (counts[s.type] !== undefined) counts[s.type]++; }); // HTML要素への反映 document.getElementById('speed-max').textContent = maxSpeed.toFixed(1); document.getElementById('speed-ave').textContent = (sumSpeed / shots.length).toFixed(1); document.getElementById('spin-max').textContent = Math.round(maxSpin); document.getElementById('spin-ave').textContent = Math.round(sumSpin); document.getElementById('count-f').textContent = counts.FOREHAND; document.getElementById('count-b').textContent = counts.BACKHAND; document.getElementById('count-s').textContent = counts.SERVE; document.getElementById('count-sm').textContent = counts.SMASH; document.getElementById('total-count').textContent = shots.length; } // 記録ボタン切替処理 function toggleRecord() { isRecording = !isRecording; if (isRecording) { sessionShots = []; // 新規セッション開始でリセット } }
